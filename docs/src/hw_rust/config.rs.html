39/******************************************************************************
40*
41*   JSON::new
42*
43******************************************************************************/
44/// Load a JSON object from a file.
45/// 
46/// # Parameters
47/// 
48/// *`p`: Path to the JSON file.
49/// 
50/// # Return
51/// 
52/// A new JSON object.
53impl JSON {
54    pub fn new(p: &PathBuf) -> Self {
55        if !p.exists() {
56            error(&format!("File {:?} does not exist!", p));
57            exit(1);
58        } // if !p.exists()
59        let contents = fs::read_to_string(p)
60            .expect(&format!("JSON::new : file {:?} cannot be read!", p));
61
62        Self {
63            value: json::parse(&contents).unwrap(),
64        } // Self
65    } // new
66} // impl

72******************************************************************************/
73/// Initialize `FILE_SYSTEM`, read `config.json` into `CONFIGURATION`, parse the
74/// CLI arguments and piped input, if there is any.
75/// 
76/// # Return
77/// 
78/// `Error` if something goes wrong.
79pub fn configure() -> Result<(), Error> {
80    let args = Args::parse();
81    FILE_SYSTEM.set(FileSystem::new())
82        .expect("`FILE_SYSTEM` initialized");
83    CONFIGURATION.set(JSON::new(&FILE_SYSTEM.get()
84        .expect(&format!("{}  Configuration file failed to load!", ERROR_PICT))
85        .config_file))
86        .expect(&format!("{}  `CONFIGURATION` could not be loaded!", ERROR_PICT));
87    ARGUMENTS.set(args).unwrap();
88    
89    if let Some(input) = get_piped_input() {
90        INPUT.set(input).unwrap();
91    }
92    Ok(())
93} // configure
