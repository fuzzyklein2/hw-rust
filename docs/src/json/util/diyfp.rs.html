14const DIY_SIGNIFICAND_SIZE: isize = 64;
15const DP_SIGNIFICAND_SIZE: isize = 52;
16const DP_EXPONENT_BIAS: isize = 0x3FF + DP_SIGNIFICAND_SIZE;
17const DP_MIN_EXPONENT: isize = -DP_EXPONENT_BIAS;
18const DP_EXPONENT_MASK: u64 = 0x7FF0000000000000;
19const DP_SIGNIFICAND_MASK: u64 = 0x000FFFFFFFFFFFFF;
20const DP_HIDDEN_BIT: u64 = 0x0010000000000000;
21
22#[derive(Copy, Clone, Debug)]
23pub struct DiyFp {
24    pub f: u64,
25    pub e: isize,
26}
27
28impl DiyFp {
29    pub fn new(f: u64, e: isize) -> Self {
30        DiyFp { f: f, e: e }
31    }
32
33    /*
34    explicit DiyFp(double d) {
35        union {
36            double d;
37            uint64_t u64;
38        } u = { d };
39
40        int biased_e = static_cast<int>((u.u64 & kDpExponentMask) >> kDpSignificandSize);
41        uint64_t significand = (u.u64 & kDpSignificandMask);
42        if (biased_e != 0) {
43            f = significand + kDpHiddenBit;
44            e = biased_e - kDpExponentBias;
45        }
46        else {
47            f = significand;
48            e = kDpMinExponent + 1;
49        }
50    }
51    */
52    pub unsafe fn from_f64(d: f64) -> Self {
53        let u: u64 = mem::transmute(d);
54
55        let biased_e = ((u & DP_EXPONENT_MASK) >> DP_SIGNIFICAND_SIZE) as isize;
56        let significand = u & DP_SIGNIFICAND_MASK;
57        if biased_e != 0 {
58            DiyFp {
59                f: significand + DP_HIDDEN_BIT,
60                e: biased_e - DP_EXPONENT_BIAS,
61            }
62        } else {
63            DiyFp {
64                f: significand,
65                e: DP_MIN_EXPONENT + 1,
66            }
67        }
68    }
69
70    /*
71    DiyFp Normalize() const {
72        DiyFp res = *this;
73        while (!(res.f & (static_cast<uint64_t>(1) << 63))) {
74            res.f <<= 1;
75            res.e--;
76        }
77        return res;
78    }
79    */
80    pub fn normalize(self) -> DiyFp {
81        let mut res = self;
82        while (res.f & (1u64 << 63)) == 0 {
83            res.f <<= 1;
84            res.e -= 1;
85        }
86        res
87    }
88
89    /*
90    DiyFp NormalizeBoundary() const {
91        DiyFp res = *this;
92        while (!(res.f & (kDpHiddenBit << 1))) {
93            res.f <<= 1;
94            res.e--;
95        }
96        res.f <<= (kDiySignificandSize - kDpSignificandSize - 2);
97        res.e = res.e - (kDiySignificandSize - kDpSignificandSize - 2);
98        return res;
99    }
100    */
101    pub fn normalize_boundary(self) -> DiyFp {
102        let mut res = self;
103        while (res.f & DP_HIDDEN_BIT << 1) == 0 {
104            res.f <<= 1;
105            res.e -= 1;
106        }
107        res.f <<= DIY_SIGNIFICAND_SIZE - DP_SIGNIFICAND_SIZE - 2;
108        res.e -= DIY_SIGNIFICAND_SIZE - DP_SIGNIFICAND_SIZE - 2;
109        res
110    }
111
112    /*
113    void NormalizedBoundaries(DiyFp* minus, DiyFp* plus) const {
114        DiyFp pl = DiyFp((f << 1) + 1, e - 1).NormalizeBoundary();
115        DiyFp mi = (f == kDpHiddenBit) ? DiyFp((f << 2) - 1, e - 2) : DiyFp((f << 1) - 1, e - 1);
116        mi.f <<= mi.e - pl.e;
117        mi.e = pl.e;
118        *plus = pl;
119        *minus = mi;
120    }
121    */
122    pub fn normalized_boundaries(self) -> (DiyFp, DiyFp) {
123        let pl = DiyFp::new((self.f << 1) + 1, self.e - 1).normalize_boundary();
124        let mut mi = if self.f == DP_HIDDEN_BIT {
125            DiyFp::new((self.f << 2) - 1, self.e - 2)
126        } else {
127            DiyFp::new((self.f << 1) - 1, self.e - 1)
128        };
129        mi.f <<= mi.e - pl.e;
130        mi.e = pl.e;
131        (mi, pl)
132    }
133}

135impl ops::Sub for DiyFp {
136    type Output = DiyFp;
137    fn sub(self, rhs: DiyFp) -> DiyFp {
138        DiyFp {
139            f: self.f - rhs.f,
140            e: self.e,
141        }
142    }
143}
144
145impl ops::Mul for DiyFp {
146    type Output = DiyFp;
147    fn mul(self, rhs: DiyFp) -> DiyFp {
148        let m32 = 0xFFFFFFFFu64;
149        let a = self.f >> 32;
150        let b = self.f & m32;
151        let c = rhs.f >> 32;
152        let d = rhs.f & m32;
153        let ac = a * c;
154        let bc = b * c;
155        let ad = a * d;
156        let bd = b * d;
157        let mut tmp = (bd >> 32) + (ad & m32) + (bc & m32);
158        tmp += 1u64 << 31; // mult_round
159        DiyFp {
160            f: ac + (ad >> 32) + (bc >> 32) + (tmp >> 32),
161            e: self.e + rhs.e + 64,
162        }
163    }
164}

242#[inline]
243pub fn get_cached_power(e: isize) -> (DiyFp, isize) {
244    let dk = (-61 - e) as f64 * 0.30102999566398114f64 + 347f64; // dk must be positive, so can do ceiling in positive
245    let mut k = dk as isize;
246    if dk - k as f64 > 0.0 {
247        k += 1;
248    }
249
250    let index = ((k >> 3) + 1) as usize;
251    let k = -(-348 + (index << 3) as isize); // decimal exponent no need lookup table
252
253    (get_cached_power_by_index(index), k)
