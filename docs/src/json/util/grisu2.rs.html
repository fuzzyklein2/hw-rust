20#[inline]
21unsafe fn grisu_round(buffer: &mut u64, delta: u64, mut rest: u64, ten_kappa: u64, wp_w: u64) {
22    while rest < wp_w && delta - rest >= ten_kappa &&
23           (rest + ten_kappa < wp_w || // closer
24            wp_w - rest > rest + ten_kappa - wp_w) {
25        *buffer -= 1;
26        rest += ten_kappa;
27    }
28}

30#[inline]
31fn count_decimal_digit32(n: u32) -> i16 {
32    if n < 10 { 1 }
33    else if n < 100 { 2 }
34    else if n < 1000 { 3 }
35    else if n < 10000 { 4 }
36    else if n < 100000 { 5 }
37    else if n < 1000000 { 6 }
38    else if n < 10000000 { 7 }
39    else if n < 100000000 { 8 }
40    // Will not reach 10 digits in digit_gen()
41    else { 9 }
42}

44#[inline]
45unsafe fn digit_gen(w: DiyFp, mp: DiyFp, mut delta: u64, mut k: i16) -> (u64, i16) {
46    static POW10: [u32; 10] = [ 1, 10, 100, 1000, 10000, 100000, 1000000, 10000000, 100000000, 1000000000 ];
47    let one = DiyFp::new(1u64 << -mp.e, mp.e);
48    let wp_w = mp - w;
49    let mut p1 = (mp.f >> -one.e) as u32;
50    let mut p2 = mp.f & (one.f - 1);
51    let mut kappa = count_decimal_digit32(p1); // kappa in [0, 9]
52
53    let mut buffer = p1 as u64;
54
55    while kappa > 0 {
56        match kappa {
57            9 => { p1 %=  100000000; }
58            8 => { p1 %=   10000000; }
59            7 => { p1 %=    1000000; }
60            6 => { p1 %=     100000; }
61            5 => { p1 %=      10000; }
62            4 => { p1 %=       1000; }
63            3 => { p1 %=        100; }
64            2 => { p1 %=         10; }
65            1 => { p1 =           0; }
66            _ => {}
67        }
68        kappa = kappa.wrapping_sub(1);
69        let tmp = ((p1 as u64) << -one.e) + p2;
70        if tmp <= delta {
71            k += kappa;
72            let pow10 = POW10[kappa as usize] as u64;
73            buffer /= pow10;
74
75            grisu_round(&mut buffer, delta, tmp, pow10 << -one.e, wp_w.f);
76            return (buffer, k);
77        }
78    }
79
80    loop {
81        p2 *= 10;
82        delta *= 10;
83        let d = (p2 >> -one.e) as u8;
84        if d != 0 || buffer != 0 {
85            buffer = buffer * 10 + d as u64;
86        }
87        p2 &= one.f - 1;
88        kappa = kappa.wrapping_sub(1);
89        if p2 < delta {
90            k += kappa;
91            let index = -(kappa as isize);
92
93            grisu_round(&mut buffer, delta, p2, one.f, wp_w.f * if index < 9 { POW10[-(kappa as isize) as usize] as u64 } else { 0 });
94            return (buffer, k);
95        }
96    }
97}

99#[inline]
100pub fn convert(float: f64) -> (u64, i16) {
101    if float == 0.0 {
102        return (0, 0);
103    }
104    unsafe {
105        let v = DiyFp::from_f64(float);
106        let (w_m, w_p) = v.normalized_boundaries();
107        let (c_mk, k) = diyfp::get_cached_power(w_p.e);
108        let w = v.normalize() * c_mk;
109        let mut wp = w_p * c_mk;
110        let mut wm = w_m * c_mk;
111        wm.f += 1;
112        wp.f -= 1;
113
114        digit_gen(w, wp, wp.f - wm.f, k as i16)
115    }
116}
